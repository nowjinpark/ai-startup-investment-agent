class ReportTooLongError(ValueError):

def _clean(value):
    if isinstance(value, list):
        return "; ".join(_clean(v) for v in value)
    if isinstance(value, dict):
        return "; ".join(f"{k}: {_clean(v)}" for k, v in value.items())
    text = str(value if value is not None else "확인 필요").replace("\x00", "")
    return re.sub(r"\s+", " ", text).strip()


def _unique_sentences(value):
    text = _clean(value)
    terms = {"contact": "도입 접점", "payer": "지불 주체", "product": "판매 대상",
             "price_unit": "과금 단위", "price_basis": "가격 산정 방식", "proven": "현장 효과 검증", "active": "사업 활동 지속"}
    pieces = re.split(r"(https?://\S+)", text)
    text = "".join(piece if piece.startswith(("http://", "https://")) else
                   re.sub(r"(?<![A-Za-z_])(" + "|".join(terms) + r")(?![A-Za-z_])", lambda match: terms[match[1]], piece)
                   for piece in pieces)
    text = text.replace("도입 접점로", "도입 접점으로")
    for translated in terms.values():
        text = text.replace(f"{translated}({translated})", translated)
    parts = re.split(r"(?<=[.!?。])\s+", text)
    return _polite_display(" ".join(dict.fromkeys(parts)))


def _polite_display(text):
    """설명문의 종결만 바꾸고 따옴표 안의 직접 인용과 URL은 보존합니다."""
    endings = {"한다": "합니다", "된다": "됩니다", "있다": "있습니다", "없다": "없습니다",
               "이다": "입니다", "했다": "했습니다", "됐다": "됐습니다"}
    protected = re.split(r'''(https?://\S+|“[^”]*”|‘[^’]*’|"[^"\n]*"|'[^'\n]*')''', text)
    for index in range(0, len(protected), 2):
        prose = re.sub(r"(?<=[가-힣A-Za-z0-9)])\s+이다(?=[.!?。](?:\s|$)|$)", "이다", protected[index])
        protected[index] = re.sub(r"(" + "|".join(endings) + r")(?=[.!?。](?:\s|$)|$)",
                                  lambda match: endings[match[1]], prose)
    return "".join(protected)


def _brief_fact(value):
    """완결된 첫 문장과 제한·계획 문장을 남겨 근거의 의미를 유지합니다."""
    sentences = re.split(r"(?<=[.!?。])\s+", _unique_sentences(value))
    if len(sentences) < 2:
        return sentences[0]
    selected = [sentences[0]]
    for sentence in sentences[1:]:
        if re.search(r"다만|그러나|미확인|확인되지|확인되지\s*않|어렵|부족|예정|계획|추정|한계|없습니다|않습니다", sentence):
            selected.append(sentence)
    return " ".join(dict.fromkeys(selected))


def _name(criterion):
    number = criterion.get("criterion_id")
    return criterion.get("name") or (NAMES[number - 1] if isinstance(number, int) and 1 <= number <= 10 else str(number))


def _number(value):
    return isinstance(value, (int, float)) and not isinstance(value, bool) and math.isfinite(value)


def _scale(record):
    return 18 if record.get("score_scale") == 18 or str(record.get("rubric_version", "")).startswith("3") else 30


def _count(record):
    return 6 if _scale(record) == 18 else 10


def _threshold(record):
    return 12 if _scale(record) == 18 else 24


def _ids(records):
    return (2, 3, 6, 7, 8, 10) if records and _scale(records[0]) == 18 else tuple(range(1, 11))


def _status(record):
    return {"pass": "추천", "conditional": "조건부 후보"}.get(record.get("decision"), "보류")


def _valid_score(record):
    return _number(record.get("total_score")) and 0 <= record["total_score"] <= _scale(record)


def _format_number(number):
    return f"{number:.1f}".removesuffix(".0")


def _score(record):
    return f"{_format_number(record['total_score'])}/{_scale(record)}" if _valid_score(record) else "미확정"


def _coverage(record):
    count = record.get("coverage_count")
    if isinstance(count, int) and not isinstance(count, bool) and 0 <= count <= _count(record):
        return count
    return sum(c.get("score") in (1, 2, 3) for c in record.get("criteria", []))


def _evidence_items(record, field):
    sources = {s["source_id"]: s for s in record.get("sources", []) if s.get("source_id")}
    valid, seen = [], set()
    for item in field.get("evidence") or []:
        source = sources.get(item.get("source_id"))
        if not source or not valid_evidence([item], [source]):
            continue
        owner = source.get("company_id")
        number = field.get("criterion_id")
        shared = owner == "__sector__" and (
            number in (1, "market_size") or str(number).startswith(("competitor_", "context_risk_"))
            or (number == 9 and field.get("_report_qualitative") is True)
        )
        discovery = owner == "__discovery__" and field.get("criterion_id") == "business" and compact(record["name"]) in compact(item["quote"])
        if owner != record["company_id"] and not shared and not discovery:
            continue
        key = (source["source_id"], compact(item["quote"]))
        if key not in seen:
            valid.append((source, _clean(item["quote"])))
            seen.add(key)
    return valid


def _checked(record):
    return [c for c in record.get("criteria", []) if c.get("score") in (1, 2, 3) and _evidence_items(record, c)]


def _analysis_checked(record):
    fields = _checked(record)
    for field in record.get("qualitative_findings") or []:
        field = {**field, "_report_qualitative": True}
        if _evidence_items(record, field):
            fields.append(field)
    technology = (record.get("details") or {}).get("technology") or {}
    used = {field.get("criterion_id") for field in fields}
    for number, value, evidence in ((4, "technology", "technology_evidence"), (5, "team", "team_evidence"),
                                     (9, "critical_risk_reason", "critical_risk_evidence")):
        field = {"criterion_id": number, "reason": technology.get(value), "evidence": technology.get(evidence) or [],
                 "_report_qualitative": True}
        if number not in used and field["reason"] and _evidence_items(record, field):
            fields.append(field)
    return fields


def _business_field(record):
    direct = record.get("overview_evidence") or []
    if direct and record.get("overview"):
        field = {"criterion_id": "business", "value": record["overview"], "evidence": direct}
        if _evidence_items(record, field):
            return field
    options = []
    for source in record.get("sources", []):
        if source.get("company_id") not in (record["company_id"], "__discovery__"):
            continue
        text = source.get("text", "")
        sentences = re.split(r"(?<=[.!?。])\s+", _clean(text))
        sentences += [line for line in text.splitlines() if line.count("|") >= 2]
        for index, sentence in enumerate(sentences):
            sentence = _clean(sentence)
            if record["name"] in sentence and sentence.count("|") < 2:
                subjects = list(re.finditer(re.escape(record["name"]) + r"(?:은|는|의|가|이)", sentence))
                begin = subjects[-1].start() if subjects else sentence.index(record["name"])
                sentence = sentence[begin:]
                if index + 1 < len(sentences) and re.match(r"실제\s*현장|해당\s*(?:제품|솔루션|시스템)", sentences[index + 1]):
                    sentence += " " + _clean(sentences[index + 1])
            if not 15 <= len(sentence) <= 300 or compact(record["name"]) not in compact(sentence):
                continue
            if re.search(r"https?://|#|저작권|무단전재|다른기사 보기|Vision|Mission|Value", sentence):
                continue
            without_name = sentence.replace(record["name"], "")
            if not re.search(r"로봇|자율|제품|서비스|제조|개발|물류|공간|스캔|재현|인식|3D|robot|autonom", without_name, re.I):
                continue
            columns = [part.strip() for part in sentence.split("|")]
            value = columns[2] if len(columns) >= 3 and compact(record["name"]) in compact(columns[1]) else sentence
            field = {"criterion_id": "business", "value": value,
                     "evidence": [{"source_id": source["source_id"], "quote": sentence}]}
            if _evidence_items(record, field):
                actions = len(re.findall(r"개발|공급|제공|판매|납품|자동화|제어|인식|재현|가공|조리|생성|설계", value))
                products = len(re.findall(r"솔루션|시스템|플랫폼|로봇|서비스|데이터|공간 MES|물류", value))
                background = len(re.findall(r"투자\s*유치|투자를|설립|분사|창업|선정|수상|대표는|대표를|전문연구인력|협\s*업", value))
                planned = len(re.findall(r"예정|계획|목표|기대", value))
                options.append((actions * 3 + products - background * 5 - planned * 2, field))
    return max(options, key=lambda item: item[0])[1] if options else None


def _market_fields(record):
    details = record.get("details") or {}
    market = details.get("market", details)
    if not isinstance(market, dict):
        return []
    value = _clean(market.get("market_size"))
    fields = []
    # 기업 매출과 성장률만으로 시장 규모를 설명하지 않습니다.
    revenue_only = "매출" in value and not re.search(r"시장\s*(?:전체\s*)?매출|시장\s*규모|설치(?:량|수|는|\s)|출하량", value)
    growth_only = "%" in value and not re.search(r"원|달러|USD|billion|million|만\s*대|천\s*대", value, re.I)
    if value not in ("확인 필요", "", "unknown") and not revenue_only and not growth_only:
        field = {"criterion_id": "market_size", "value": value, "evidence": market.get("market_size_evidence") or []}
        if _evidence_items(record, field):
            fields.append(field)
    for index, competitor in enumerate(market.get("competitors") or []):
        if not isinstance(competitor, dict) or not competitor.get("name"):
            continue
        field = {"criterion_id": f"competitor_{index}", "value": competitor["name"], "comparison": competitor.get("comparison", ""),
                 "evidence": competitor.get("evidence") or []}
        evidence = _evidence_items(record, field)
        if evidence and compact(field["value"]).casefold() in compact(" ".join(quote for _, quote in evidence)).casefold():
            fields.append(field)
    for index, risk in enumerate(market.get("context_risks") or []):
        if not isinstance(risk, dict) or not risk.get("reason"):
            continue
        field = {"criterion_id": f"context_risk_{index}", "value": risk["reason"],
                 "risk_type": _clean(risk.get("type") or "업계"), "evidence": risk.get("evidence") or []}
        if _evidence_items(record, field):
            fields.append(field)
    return fields


def _references(records):
    refs, by_url, lookup = [], {}, {}
    for record in records:
        business = _business_field(record)
        for field in _analysis_checked(record) + _market_fields(record) + ([business] if business else []):
            numbers = []
            for source, quote in _evidence_items(record, field):
                url = str(source.get("url") or "")
                key = url or source["source_id"]
                if key not in by_url:
                    organization = source.get("publisher") or source.get("author") or source.get("organization") or urlsplit(url).netloc.removeprefix("www.") or "작성기관 미확인"
                    ref = {"number": len(refs) + 1, "title": source.get("title") or "제목 미기재", "url": url, "organization": _clean(organization),
                           "pages": set(), "published": set(), "collected": set(), "quotes": []}
                    refs.append(ref)
                    by_url[key] = ref
                ref = by_url[key]
                if source.get("original_page") is not None:
                    ref["pages"].add(str(source["original_page"]))
                for original, label in [("published_at", "published"), ("collected_at", "collected")]:
                    if source.get(original):
                        ref[label].add(str(source[original])[:10])
                if quote not in ref["quotes"]:
                    ref["quotes"].append(quote)
                if ref["number"] not in numbers:
                    numbers.append(ref["number"])
            lookup[(record["company_id"], field["criterion_id"])] = numbers
    return refs, lookup


def _cite(record, field, lookup):
    numbers = lookup.get((record["company_id"], field.get("criterion_id")), [])
    return " [" + ",".join(map(str, numbers)) + "]" if numbers else ""


def _company_cites(record, lookup):
    numbers = sorted({number for (company, _), ids in lookup.items() if company == record["company_id"] for number in ids})
    return " [" + ",".join(map(str, numbers)) + "]" if numbers else ""


def _reason_lines(record):
    reasons = [_unique_sentences(r) for r in record.get("reasons") or [] if _clean(r)]
    if record.get("decision") != "pass" and not reasons:
        reasons = ["보류 사유를 확인해야 합니다."]
    return list(dict.fromkeys(reasons))


def _brief_reasons(record):
    """반복되는 평가 조건만 묶고 개별 기업의 실질적인 사유는 유지합니다."""
    qualifications, missing, conditions, other = [], [], [], []
    for reason in _reason_lines(record):
        match = re.fullmatch(r"기업 자격 미확인: (.+?) — 출처와 원문 인용이 없습니다\.", reason)
        if match:
            qualifications.append(match[1])
            continue
        match = re.fullmatch(r"기업 자격 (.+): 출처와 원문 인용이 없습니다\.", reason)
        if match:
            qualifications.append(match[1])
            continue
        match = re.fullmatch(r"(.+): 출처와 원문 인용이 없습니다\.", reason)
        if match:
            missing.append(match[1])
            continue
        if reason == "중대한 미해결 위험 여부와 원문 근거를 확인해야 합니다.":
            missing.append("중대위험 여부")
            continue
        match = re.fullmatch(r"확인된 평가 항목이 (\d+)/6개입니다\. 모든 항목의 근거가 있어야 총점을 계산합니다\.", reason)
        if match:
            conditions.append(f"평가 근거 {match[1]}/6개 확인으로 총점 미확정입니다.")
            continue
        match = re.fullmatch(r"확인된 평가 항목이 (\d+)/10개로 최소 (\d+)개에 미달합니다\.", reason)
        if match:
            conditions.append(f"확인 항목 {match[1]}/10개(최소 {match[2]}개 미달)")
            continue
        match = re.fullmatch(r"필수 평가 항목 미확인: (.+)번", reason)
        if match:
            conditions.append(f"필수 {match[1]}번 미확인")
            continue
        match = re.fullmatch(r"확인 항목 환산점수 ([\d.]+)점으로 통과 기준 ([\d.]+)점에 미달합니다\.", reason)
        if match:
            conditions.append(f"환산 {match[1]}점 < 기준 {match[2]}점")
            continue
        match = re.fullmatch(r"미확인 항목을 고려한 점수 범위의 하한 ([\d.]+)점이 통과 기준 ([\d.]+)점에 미달합니다\.", reason)
        if match:
            conditions.append(f"보수적 하한 {match[1]}점 < 기준 {match[2]}점")
            continue
        other.append(reason)
    grouped = []
    if qualifications:
        grouped.append("자격 근거 미확인: " + "·".join(qualifications))
    if missing:
        grouped.append("원문 근거 미확인: " + "·".join(missing))
    if conditions:
        grouped.append("; ".join(conditions))
    return grouped + other


def _missing(record):
    return [_name(c) for c in record.get("criteria", []) if c.get("score") is None]


def _excluded_count(metadata):
    for key in ("excluded_count", "rejected_count"):
        if isinstance(metadata.get(key), int):
            return metadata[key]
    return len(metadata.get("excluded") or metadata.get("excluded_candidates") or metadata.get("rejected_candidates") or [])


def _summary(records, passed, metadata, lookup, compact_mode=False):
    if records and _scale(records[0]) == 18:
        qualified = [r for r in records if r.get("decision") == "pass"]
        conditional = [r for r in records if r.get("decision") == "conditional"]
        score_pass = sum(r.get("score_pass", _valid_score(r) and r["total_score"] >= 12) for r in records)
        candidate_count = metadata.get("candidate_count")
        scope = (f"국내 Physical AI 후보 {candidate_count}개 중 {len(records)}개 기업을"
                 if isinstance(candidate_count, int) and candidate_count > len(records)
                 else f"국내 Physical AI 후보 {len(records)}개 기업을")
        lines = [f"{scope} 6개 항목으로 평가했습니다. 18점 중 12점 이상을 충족한 기업은 {score_pass}개이며, 자격 확인을 마친 추천 기업은 {len(qualified)}개, 자격 확인이 필요한 조건부 후보는 {len(conditional)}개입니다."]
        if isinstance(candidate_count, int) and candidate_count > len(records):
            prefix = "후보 5개 확보로 평가를 종료했으며, " if metadata.get("stop_reason") == "five_candidates" else ""
            lines.append(f"{prefix}나머지 {candidate_count - len(records)}개 후보는 미평가 상태입니다.")
        if passed:
            leader = passed[0]
            if not compact_mode:
                lines.append(f"점수 기준 최상위 기업은 {leader['name']}({_score(leader)}, {_status(leader)})입니다.")
            for group, label in ((qualified, "추천"), (conditional, "조건부 후보")):
                if group:
                    names = ", ".join(f"{r['name']}({_score(r)})" for r in sorted(group, key=lambda r: -r["total_score"]))
                    lines.append(f"{label}: {names}입니다.")
            checked = _checked(leader)
            strengths = sorted(checked, key=lambda c: ({3: 0, 2: 1, 10: 2}.get(c.get("criterion_id"), 3), -c["score"]))[:2]
            for criterion in strengths:
                reason = _brief_fact(criterion.get("reason")) if compact_mode else _unique_sentences(criterion.get("reason"))
                lines.append(f"주요 근거 · {_name(criterion)}: {reason}{_cite(leader, criterion, lookup)}")
        if conditional:
            lines.append("조건부 후보는 비상장·투자 단계·Exit 등 기업 자격의 추가 확인이 필요합니다.")
        held = [r for r in records if r.get("decision") == "hold"]
        if held:
            unknown = sum(not _valid_score(r) for r in held)
            below = sum(_valid_score(r) and r["total_score"] < 12 for r in held)
            counts = Counter(name for r in held for name in _missing(r))
            line = f"보류 사유: {len(held)}개 중 총점 미확정 {unknown}개, 12점 미만 {below}개입니다."
            if counts:
                line += " 주요 미확인 항목은 " + ", ".join(f"{name}({count}개)" for name, count in counts.most_common(2)) + "입니다."
            lines.append(line)
            blocked = [r for r in held if r.get("score_pass")]
            if blocked:
                lines.append(f"점수 기준을 충족한 {len(blocked)}개 기업은 기업 자격·위험 검토 조건을 충족하지 못해 보류되었습니다.")
        excluded = _excluded_count(metadata)
        if excluded:
            lines.append(f"탐색 단계에서 제외된 {excluded}개 기업은 평가 건수에 포함하지 않았습니다.")
        return lines

    if not records:
        lines = ["평가가 완료된 기업이 없어 투자 검토 후보를 선정하지 못했습니다."]
    else:
        lines = [f"국내 Physical AI 후보 {len(records)}개 기업을 평가하여 {len(passed)}개를 투자 검토 후보로 선정했습니다."]
        if passed:
            leader = passed[0]
            lines[0] += f" 최상위 후보는 {leader['name']}({_score(leader)}, {_coverage(leader)}/10개 항목 확인)입니다."
            strengths = [c for c in _checked(leader) if c.get("score") == 3][:2]
            for criterion in strengths:
                reason = f"{_name(criterion)} 상 등급" if compact_mode else _unique_sentences(criterion.get("reason"))
                lines.append(f"주요 근거 · {_name(criterion)}: {reason}{_cite(leader, criterion, lookup)}")
            weak = next((c for c in _checked(leader) if c.get("score") in (1, 2)), None)
            if weak:
                reason = f"{_name(weak)} 추가 확인" if compact_mode else _unique_sentences(weak.get("reason"))
                lines.append(f"보완 항목 · {_name(weak)}: {reason}{_cite(leader, weak, lookup)}")
        else:
            unscored = sum(not _valid_score(record) for record in records)
            under = sum(_valid_score(record) and record["total_score"] < 24 for record in records)
            if unscored == len(records):
                lines.append(f"{unscored}개 기업 모두 필수 평가 근거가 부족해 총점을 확정하지 못했습니다. 미확정 기업을 최하위 기업으로 간주하지 않습니다.")
            else:
                lines.append(f"총점 미확정 {unscored}개, 확인된 환산점수 24점 미만 {under}개입니다. 미확정 기업을 최하위 기업으로 간주하지 않습니다.")
            counts = Counter(name for record in records for name in _missing(record))
            for name, count in counts.most_common(2):
                lines.append(f"주요 미확인 항목 · {name}: {count}개 기업에서 근거를 확인하지 못했습니다.")
            if not counts:
                reasons = Counter(reason for record in records for reason in _reason_lines(record))
                if reasons:
                    reason, count = reasons.most_common(1)[0]
                    lines.append(f"주요 보류 사유 · {count}개 기업: {reason}")
    excluded = _excluded_count(metadata)
    if excluded:
        lines.append(f"탐색 단계에서 제외된 {excluded}개 기업은 위 평가 건수와 순위에 포함하지 않았습니다.")
    unresolved = len(metadata.get("unresolved") or metadata.get("unresolved_candidates") or [])
    if unresolved:
        lines.append(f"기업 자격 확인을 기다리는 후보는 별도로 {unresolved}개입니다.")
    return lines


def _observations(records):
    if not records:
        return []
    counts = Counter(name for record in records for name in _missing(record))
    sentences = [f"이번 평가 표본에서 ‘{name}’ 근거가 부족한 기업은 {count}개입니다." for name, count in counts.most_common(2)]
    if not sentences:
        weak = Counter(_name(c) for record in records for c in record.get("criteria", []) if c.get("score") in (1, 2))
        sentences = [f"이번 평가 표본에서 ‘{name}’이 중·하 등급인 기업은 {count}개입니다." for name, count in weak.most_common(2)]
    return sentences


def _followup_review(records):
    if records and _scale(records[0]) == 18:
        comparable = [r for r in records if _valid_score(r)]
        if comparable:
            company = max(comparable, key=lambda r: r["total_score"])
            return f"12점 기준선에 가장 가까운 기업은 {company['name']}({_score(company)})입니다. 보류 사유: " + " / ".join(_reason_lines(company))
        available = [r for r in records if _coverage(r)]
        if available:
            company = max(available, key=_coverage)
            return f"추가 확인 우선 기업은 {company['name']}({_coverage(company)}/6개 항목 확인)입니다. 총점 미확정 상태이므로 투자 점수 순위를 뜻하지 않습니다."
        return "확인된 평가 항목이 없어 기준선에 가장 가까운 기업을 판단할 수 없습니다."

    comparable = [record for record in records if _valid_score(record)]
    if comparable:
        company = max(comparable, key=lambda record: record["total_score"])
        if company["total_score"] >= 24:
            text = f"{company['name']}의 환산점수는 {_score(company)}로 24점 기준을 충족했으나 투자 검토는 보류되었습니다."
            bounds = company.get("score_range") or {}
            if _number(bounds.get("min")) and bounds["min"] < 24:
                text += f" 미확인 항목을 반영한 보수적 하한은 {_format_number(bounds['min'])}점으로 24점에 미달합니다."
            return text + " 보류 사유: " + " / ".join(_reason_lines(company))
        text = f"환산점수가 확인된 기업 중 24점 기준선에 가장 가까운 기업은 {company['name']}({_score(company)}, {_coverage(company)}/10 확인)입니다."
        needs = [_name(criterion) for criterion in company.get("criteria", []) if criterion.get("score") != 3]
        if needs:
            text += " 추가 검토 항목은 " + ", ".join(needs) + "입니다."
        return text
    available = [record for record in records if _coverage(record) > 0]
    if not available:
        return "총점과 확인된 평가 항목이 없어 기준선에 가장 가까운 기업을 판단할 수 없습니다. 기업별 미확인 사유를 비교합니다."
    company = max(available, key=_coverage)
    text = f"추가 확인 우선 기업은 {company['name']}입니다. 확인 항목이 가장 많은 기업 중 하나({_coverage(company)}/10)이며, 총점 미확정 상태이므로 투자 점수 순위를 뜻하지 않습니다."
    criteria = {criterion.get("criterion_id"): criterion for criterion in company.get("criteria", [])}
    missing_core = [f"{number}번 {NAMES[number - 1]}" for number in (2, 3, 4, 9) if criteria.get(number, {}).get("score") not in (1, 2, 3)]
    if missing_core:
        text += " 부족한 필수 항목은 " + ", ".join(missing_core) + "입니다."
    else:
        missing = _missing(company)
        if missing:
            text += " 추가 확인 항목은 " + ", ".join(missing) + "입니다."
    return text

def _styles(size):
    base = dict(fontName="Report", fontSize=size, leading=size * 1.35, wordWrap="CJK", splitLongWords=True,
                textColor=colors.black, alignment=TA_LEFT, spaceAfter=4 if size >= 9.5 else 2)
    body = ParagraphStyle("body", **base)
    table = ParagraphStyle("table", **{**base, "spaceAfter": 0, "leading": size * 1.25})
    heading = ParagraphStyle("heading", fontName="ReportBold", fontSize=12, leading=16, spaceBefore=9, spaceAfter=6, keepWithNext=True)
    title = ParagraphStyle("title", fontName="ReportBold", fontSize=17, leading=22, spaceAfter=9, keepWithNext=True)
    company = ParagraphStyle("company", fontName="ReportBold", fontSize=size + 1, leading=size * 1.4, spaceBefore=7 if size >= 9.5 else 5, spaceAfter=3, keepWithNext=True)
    return {"body": body, "table": table, "heading": heading, "title": title, "company": company}


def _paragraph(text, style, markup=False):
    return Paragraph(text if markup else escape(_clean(text)), style)


def _table(rows, widths, style, header=True):
    cells = [[_paragraph(value, style) for value in row] for row in rows]
    table = LongTable(cells, colWidths=widths, repeatRows=1 if header else 0, splitByRow=1, splitInRow=1, hAlign="LEFT")
    commands = [("VALIGN", (0, 0), (-1, -1), "TOP"), ("LEFTPADDING", (0, 0), (-1, -1), 4),
                ("RIGHTPADDING", (0, 0), (-1, -1), 4), ("TOPPADDING", (0, 0), (-1, -1), 4),
                ("BOTTOMPADDING", (0, 0), (-1, -1), 4), ("LINEBELOW", (0, 0), (-1, 0), .6, colors.black)]
    if header:
        commands += [("BACKGROUND", (0, 0), (-1, 0), colors.HexColor("#eeeeee"))]
    table.setStyle(TableStyle(commands))
    return table

def _reference_meta(reference):
    items = ["원문 페이지 " + (", ".join(sorted(reference["pages"], key=lambda value: (not value.isdigit(), int(value) if value.isdigit() else value))) or "웹 본문")]
    for key, label in [("published", "발행"), ("collected", "수집")]:
        if reference[key]:
            items.append(label + " " + ", ".join(sorted(reference[key])))
    return " · ".join(items)


def _reference_html(reference):
    date = ", ".join(sorted(reference["published"])) or "발행일 미확인"
    display_title = _clean(reference["title"]).replace("日", "일본").replace("美", "미국").replace("‧", "·").replace("∙", "·")
    title = f"[{reference['number']}] {reference['organization']} ({date}). {display_title}."
    if reference["pages"]:
        pages = sorted(reference["pages"], key=lambda value: (not value.isdigit(), int(value) if value.isdigit() else value))
        title += " 원문 페이지 " + ", ".join(pages) + "."
    if reference["collected"]:
        title += " 수집 " + ", ".join(sorted(reference["collected"])) + "."
    url = reference["url"]
    visible = unquote(url) if url else "원문 URL 미기재"
    link = f'<link href="{escape(url, quote=True)}" color="black">{escape(visible)}</link>' if url.startswith(("https://", "http://")) else escape(visible)
    return f"{escape(title)}<br/>{link}"


def _detail_records(records, passed, profile):
    if profile < 2:
        return passed + [r for r in records if r not in passed]
    return passed or sorted(records, key=lambda record: -_coverage(record))[:3 if profile >= 4 else 5]


def _hold_facts(record):
    checked = _checked(record)
    first = next((c for c in checked if c.get("criterion_id") in (2, 4)), next(iter(checked), None))
    risk = next((c for c in checked if c.get("criterion_id") == 9), None)
    return [c for index, c in enumerate([first, risk]) if c and c not in [first, risk][:index]]


def _held_fields(record):
    fields = _hold_facts(record) + _market_fields(record)
    fields += [f for f in _analysis_checked(record) if f.get("criterion_id") == 9 and f not in fields]
    return fields

        values.append(value + _cite(record, field, lookup))
    unknown = [str(c["criterion_id"]) for c in record.get("criteria", []) if c.get("score") is None]
    if unknown:
        values.append("미확인 평가항목: " + "·".join(unknown) + ".")
    reasons = _brief_reasons(record)
    if reasons:
        values.append("보류 사유: " + " / ".join(reasons))
    return " ".join(values)


def _pdf_references(records, passed, refs, lookup, profile):
    if profile == 0:
        return refs, lookup
    detailed_ids = {r["company_id"] for r in _detail_records(records, passed, profile)}
    keys = set()
    for record in records:
        if record["company_id"] not in detailed_ids:
            fields = _held_fields(record) if profile >= 4 else _hold_facts(record)
        elif profile >= 4:
            fields = _focused_fields(record)
        else:
            fields = []
            for group in ((2, 4, 10), (5,), (1, 3, 6, 7, 8), (9,)):
                fields += [c for c in _analysis_checked(record) if c.get("criterion_id") in group][:2 if profile == 1 else 1]
            fields += _market_fields(record)
            business = _business_field(record)
            if business:
                fields.append(business)
        keys.update((record["company_id"], field["criterion_id"]) for field in fields)
    if passed:
        leader = passed[0]
        fields = (sorted(_checked(leader), key=lambda c: ({3: 0, 2: 1, 10: 2}.get(c.get("criterion_id"), 3), -c["score"]))[:2]
                  if _scale(leader) == 18 else [c for c in _checked(leader) if c.get("score") == 3][:2])
        fields += [c for c in _checked(leader) if c.get("score") in (1, 2)][:1]
        keys.update((leader["company_id"], field["criterion_id"]) for field in fields)
    selected_lookup = {key: numbers for key, numbers in lookup.items() if key in keys}
    used = {n for numbers in selected_lookup.values() for n in numbers}
    return [reference for reference in refs if reference["number"] in used], selected_lookup

def _markdown(records, passed, metadata, refs, lookup, observations):
    lines = ["# 국내 Physical AI 스타트업 투자 검토 보고서", ""]
    if metadata.get("demo"):
        lines += ["**테스트 자료 · 실제 기업에 대한 투자 판단이 아닙니다.**", ""]
    lines += ["## SUMMARY", "", *_summary(records, passed, metadata, lookup), "", "## 기업별 평가", ""]
    ranked = passed + [r for r in records if r not in passed]
    for record in ranked:
        lines.append(f"- {record['name']}: {_score(record)}, {_coverage(record)}/{_count(record)} 확인, {_status(record)}")
    if records and not passed:
        lines += ["", "## 추가 확인 대상", "", _followup_review(records)]
    for record in ranked:
        lines += ["", f"## {record['name']}", "", f"점수: {_score(record)} · 확인 항목: {_coverage(record)}/{_count(record)}", ""]
        if record.get("score_range"):
            lines.append("점수 범위: " + _clean(record["score_range"]))
        for label, value in _company_sections(record, lookup, 0):
            lines.append(f"- {label}: {value}")
        lines += ["", "### 항목별 평가", ""]
        for criterion in record.get("criteria", []):
            score = criterion.get("score")
            shown = f"{score}점" if score in (1, 2, 3) else "미확인"
            reason = _unique_sentences(criterion.get("reason") or "근거 확인 필요")
            prefix = "판정 근거" if criterion in _checked(record) else "확인 과제"
            lines.append(f"- {criterion.get('criterion_id')}. {_name(criterion)}: {shown} / {prefix}: {reason}{_cite(record, criterion, lookup)}")
    if not passed and observations:
        lines += ["", "## 평가 표본의 시사점", "", *observations, "이번에 평가한 기업과 수집 자료에 한정한 결과입니다."]
    lines += ["", "## REFERENCE", ""]
    for reference in refs:
        date = ", ".join(sorted(reference["published"])) or "발행일 미확인"
        lines += [f"[{reference['number']}] {reference['organization']} ({date}). {reference['title']}.", _reference_meta(reference), reference["url"] or "원문 URL 미기재",
                  *["> " + quote for quote in reference["quotes"]], ""]
    return "\n".join(lines)


def create_report(records, output_dir, metadata, writer=None):
    """점수·판정을 유지하고 근거, 보류 사유, 출처를 자르지 않습니다."""
    records, metadata = deepcopy(records), deepcopy(metadata)
    if len(records) > 20 or len({r.get("company_id") for r in records}) != len(records):
        raise ValueError("평가 기업은 중복 없이 최대 20개입니다.")
    if len({_scale(record) for record in records}) > 1:
        raise ValueError("서로 다른 평가표의 점수는 하나의 보고서에서 비교할 수 없습니다.")
    if any(r.get("decision") in ("pass", "conditional") and (not _valid_score(r) or r["total_score"] < _threshold(r)) for r in records):
        raise ValueError("후보 기업에는 해당 평가표의 통과 기준 이상인 확인된 평가 점수가 필요합니다.")
    for record in records:
        if _scale(record) == 18 and record.get("decision") in ("pass", "conditional"):
            expected = "eligible" if record["decision"] == "pass" else "pending"
            if record.get("eligibility_status") != expected:
                raise ValueError("추천과 조건부 후보는 기업 자격 확인 상태에 맞게 구분해야 합니다.")
    passed = sorted([r for r in records if r.get("decision") in ("pass", "conditional")], key=lambda r: (-r["total_score"], r["name"]))[:5]
    refs, lookup = _references(records)
    # 보고서 단계는 추가 API 호출 없이 확인된 결과만 정리합니다.
    observations = _observations(records)
    output_dir = Path(output_dir)
    output_dir.mkdir(parents=True, exist_ok=True)
    pdf_path, markdown_path = output_dir / "investment-report.pdf", output_dir / "investment-report.md"
    markdown_path.write_text(_markdown(records, passed, metadata, refs, lookup, observations), encoding="utf-8")
    _register_fonts()
    # 작은 글씨나 누락 대신 중복 설명과 보조 분석의 양부터 줄입니다.
    attempts = ([(9.5, 4, True), (9, 4, True), (8.5, 4, True)]
                if records and _scale(records[0]) == 18 else
                [(9.5, 0, False), (9, 1, False), (9, 3, True), (8.5, 3, True)])
    last_pages = None
    for size, profile, short_summary in attempts:
        buffer = BytesIO()
        document = _Document(buffer, pagesize=A4, leftMargin=MARGIN, rightMargin=MARGIN,
                             topMargin=MARGIN, bottomMargin=34, title="국내 Physical AI 스타트업 투자 검토 보고서", author="울산 1반")
        document.summary_fits = False
        pdf_refs, pdf_lookup = _pdf_references(records, passed, refs, lookup, profile)
        document.build(_story(records, passed, metadata, pdf_refs, pdf_lookup, observations, size, profile, short_summary),
                       onFirstPage=_footer, onLaterPages=_footer)
        payload = buffer.getvalue()
        last_pages = len(PdfReader(BytesIO(payload)).pages)
        if last_pages <= 5 and document.summary_fits:
            pdf_path.write_bytes(payload)
            result = {"pdf_path": str(pdf_path.resolve()), "markdown_path": str(markdown_path.resolve()),
                      "page_count": last_pages, "body_font_size": size, "layout_profile": profile}
            return result
    raise ReportTooLongError(f"핵심 보류 사유와 전체 출처를 보존하면 PDF {last_pages}쪽으로 5쪽을 초과하거나 SUMMARY가 반쪽을 넘습니다. 전체 내용은 {markdown_path.resolve()}에 저장했습니다. 중복 원문과 보고서 범위를 검토해 주세요.")
