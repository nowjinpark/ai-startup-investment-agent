# 03. 시장경쟁분석 · 김희윤

시장·고객·과금·경쟁의 검색 질문과 근거 해석을 수정합니다.

수정할 파일은 `agents/market_competition.py`입니다. 연결 규칙은 `CONTRACT.md`, 평가표는 `config/rubric.json`에 있습니다. 아래는 전달용 코드 사본이며, 실제 실행에는 전체 프로젝트와 Python 파일이 필요합니다.

## 담당 부분 실행

설치 후 가상 예제를 한 번 실행합니다.

```bash
./.venv/bin/python main.py demo --output output/demo
./.venv/bin/python run_agent.py market --demo --run output/demo --index 0
```

실제 저장 자료는 `--demo`를 빼고 `--run 실제_실행_결과_폴더`로 지정합니다. 탐색은 저장된 준비 자료를 기본 재사용하며 이때 API를 호출하지 않습니다. 새 수집(`--refresh-data`)·기술·시장 분석과 투자 판단 전 원문 감사에는 API 비용이 발생합니다. 점수 계산 자체와 보고서 재생성은 무료입니다. 결과는 `part-market/`에 저장됩니다. 각자 수정한 분석 결과를 통합할 때는 `company_id`·`round_id`·평가기준 버전을 맞춥니다.

## 코드

```python
import json

from common import ROOT, review_findings, review_details
from models import MarketOutput

CRITERION_QUESTIONS = {
    3: '유료 PoC 계약 공급 납품 체결 날짜 고객사 발표',
    6: '실제 도입 고객 현장 검증 PoC 고객사 이름과 도입 현장',
    7: '지불 주체 판매 제품 가격 단위 가격 산정 근거 렌탈 구독 판매',
    8: '동일 기술 적용 산업 사용처 확장 실제 검증 사례 경쟁 제품 성능 가격',
}
CRITERIA = list(CRITERION_QUESTIONS)
QUESTIONS = [
    '제품이 타깃하는 세부시장 규모 고객 수요 조사 지역 기준 연도',
    *CRITERION_QUESTIONS.values(),
]


def _unknown_result(company_id, round_id, rubric_version):
    return {
        'company_id': company_id, 'round_id': round_id,
        'status': 'insufficient', 'rubric_version': rubric_version,
        'summary': '시장·고객 자료 미확인',
        'findings': [
            {'criterion_id': number, 'status': 'unknown', 'numeric_value': None,
             'category_value': None, 'items': [], 'evidence': [],
             'reason': '시장·고객 자료를 검색했으나 평가 근거를 찾지 못했습니다.'}
            for number in CRITERIA
        ],
        'sources': [],
        'details': {'market_size': '확인 필요', 'market_size_evidence': [],
                    'market_scope': '', 'cagr_period': '',
                    'business_model': '', 'competitors': []},
        'risks': [], 'missing': ['시장·고객 자료'], 'error': None,
    }


def run(runtime, candidate, round_id):
    cid = candidate['company_id']
    chunks = runtime.corpus.evidence_for([candidate['name'] + ' ' + q for q in QUESTIONS], cid, 'market')
    # 공통 산업 근거를 기업명 검색 순위에 종속시키지 않습니다. 기업 실적 채점에는 사용할 수 없습니다.
    sector_queries = [candidate.get('overview', '') + ' 대상 제품 시장 규모 지역 연도 수요',
                      'Physical AI 로봇 시장 규제 안전 운영 위험 대안 기술 경쟁 제품']
    sector = runtime.corpus.evidence_for(sector_queries, None, 'market')
    chunks = list({c.get('chunk_id', c['source_id']): c for c in chunks + sector}.values())
    sources = runtime.corpus.source_pages(chunks)
    rubric = json.loads((ROOT / 'config/rubric.json').read_text(encoding='utf-8'))
    if not chunks:
        return _unknown_result(cid, round_id, rubric['version'])
    rules = [r for r in rubric['criteria'] if r['id'] in CRITERIA]
    result = runtime.ask(MarketOutput,
        '시장·경쟁 분석 담당입니다. 기술 분석 결과 없이 현재 기업의 시장·고객 근거를 분석합니다. '
        'findings에는 3,6,7,8번의 사실만 추출합니다. payment_stage는 paid/contact/none입니다. '
        'MOU·LOI를 유료 계약으로 간주하지 않습니다. 공공조달 기록 부재는 매출 부재가 아닙니다. '
        '같은 고객의 사업장은 고객 한 곳, 재인용 뉴스는 같은 거래입니다. '
        '6번은 실제 검증 고객·현장 수이고, 8번은 구분되는 적용 산업·사용처 수입니다. '
        '계획·가능성을 검증 완료된 산업이나 고객으로 세지 않습니다. 반복 기사와 같은 현장은 중복 집계하지 않습니다. '
        'pricing_components는 payer/product/price_unit/price_basis입니다. '
        '경쟁사와 대상 기업의 실적을 구분합니다. 시장 규모와 경쟁 비교는 점수와 별개의 정성 설명입니다. '
        '성장률은 채점에서 제외했으므로 cagr_period는 빈 문자열로 둡니다. '
        '시장 규모는 market_size_evidence에 원문과 출처를 붙이고 없으면 확인 필요입니다. '
        '경쟁 비교는 공통 산업자료도 사용해 같은 문제를 해결하는 대안과 대상기업의 각각 확인된 사실을 각 출처와 함께 나란히 설명할 수 있습니다. '
        '원문이 직접 비교한 경우 comparison_kind=direct, 각사 사실을 나란히 설명하면 alternative이며 두 개 이상의 서로 다른 원문 출처가 필요합니다. '
        '직접 비교 시험이 없으면 더 우수하다·더 저렴하다 같은 상대적 성능이나 가격 우위를 추론하지 않습니다. '
        '유료 근거가 없더라도 MOU·실증·PoC가 확인되면 3번은 contact입니다. '
        '고객 현장 1곳이면 6번의 유효한 근거이며 복수 미확인을 이유로 버리지 않습니다. '
        '7번은 확인된 과금 요소만 items에 남깁니다. 2개 확인은 유효하며 나머지 미공개를 이유로 전체를 unknown으로 만들지 않습니다. '
        '기업 매출은 시장 규모가 아닙니다. 성장률만 확인되면 시장 규모는 확인 필요입니다. '
        '경쟁 비교도 근거가 없으면 확인 필요입니다. 일반 산업자료는 기업별 유료실적 근거가 아닙니다. '
        '공개되지 않은 숫자를 0으로 채우지 않습니다. 각 항목은 실제 인용과 반환합니다.',
        {'company': candidate['name'], 'overview': candidate.get('overview', ''), 'rules': rules, 'sources': chunks})
    result['findings'] = review_findings(runtime, candidate, [f for f in result['findings'] if f['criterion_id'] in CRITERIA], sources,
        {'market_scope': result['market_scope'], 'cagr_period': result['cagr_period']})
    claims = [{'criterion_id': 12, 'claim': {'market_size': result['market_size'], 'scope': result['market_scope']},
               'evidence': result['market_size_evidence']}]
    claims.extend({'criterion_id': 20+i, 'claim': {'competitor': c['name'], 'comparison': c['comparison'], 'comparison_kind': c.get('comparison_kind', 'direct')},
                   'evidence': c['evidence']} for i, c in enumerate(result['competitors']))
    approved = review_details(runtime, candidate, claims, sources)
    if 12 not in approved:
        result['market_size'] = '확인 필요'
        result['market_size_evidence'] = []
    result['competitors'] = [c for i,c in enumerate(result['competitors']) if 20+i in approved]
    return {'company_id': cid, 'round_id': round_id, 'status': 'completed',
            'rubric_version': rubric['version'],
            'summary': result['summary'], 'findings': result['findings'], 'sources': sources,
            'details': {k: result[k] for k in ['market_size', 'market_size_evidence', 'market_scope', 'cagr_period', 'business_model', 'competitors']},
            'risks': result['risks'], 'missing': result['missing'], 'error': None}
```
