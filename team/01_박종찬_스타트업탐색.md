# 01. 스타트업탐색 · 박종찬

후보 탐색·자격 확인과 원문 수집을 수정합니다.

수정할 파일은 `agents/discovery.py`입니다. 연결 규칙은 `CONTRACT.md`, 평가표는 `config/rubric.json`에 있습니다. 아래는 전달용 코드 사본이며, 실제 실행에는 전체 프로젝트와 Python 파일이 필요합니다.

## 담당 부분 실행

설치 후 가상 예제를 한 번 실행합니다.

```bash
./.venv/bin/python main.py demo --output output/demo
./.venv/bin/python run_agent.py discovery --demo --run output/demo --index 0
```

실제 저장 자료는 `--demo`를 빼고 `--run 실제_실행_결과_폴더`로 지정합니다. 탐색은 저장된 준비 자료를 기본 재사용하며 이때 API를 호출하지 않습니다. 새 수집(`--refresh-data`)·기술·시장 분석과 투자 판단 전 원문 감사에는 API 비용이 발생합니다. 점수 계산 자체와 보고서 재생성은 무료입니다. 결과는 `part-discovery/`에 저장됩니다. 각자 수정한 분석 결과를 통합할 때는 `company_id`·`round_id`·평가기준 버전을 맞춥니다.

```python
import hashlib
import json
import re
import time
from datetime import datetime, timezone
from urllib.parse import urlsplit

from collection import Collector
from common import ROOT, compact, save_json, valid_evidence
from models import CandidateHints, EligibilityResult, EvidenceReview
from rag import Corpus


DOMAIN_WORDS = re.compile(r'로봇|robot|자율\s*주행|자율\s*운항|무인\s*(이동|운반|항공)|드론|physical\s*ai|피지컬\s*ai', re.I)
QUALIFICATION_KEYS = ['identity', 'domestic', 'unlisted', 'stage', 'no_exit', 'physical_ai', 'operating']


def company_id(name):
    normalized = re.sub(r'주식회사|\(주\)|[^가-힣a-z0-9]', '', name.casefold())
    return 'co-' + hashlib.sha256(normalized.encode()).hexdigest()[:12]


def _mentions_company(quote, names):
    normalized = compact(quote).casefold()
    return any(compact(name).casefold() in normalized for name in names if name.strip())


def _unknown(reason):
    return {'status': 'unknown', 'reason': reason, 'evidence': []}


def _unknown_eligibility(reason):
    return {key: _unknown(reason) for key in QUALIFICATION_KEYS[1:6]}


def _diagnostics(runtime):
    save_json(runtime.output_dir / 'discovery_diagnostics.json', runtime.discovery)


def _within_time(runtime):
    if time.monotonic() - runtime.preparation_started < runtime.config.get('preparation_max_seconds', 1800):
        return True
    runtime.discovery['stop_reason'] = 'preparation_time_limit'
    return False


def _collect_url(runtime, url, owner, role, remaining, title=None, pages=None):
    if remaining <= 0 or not _within_time(runtime):
        return []
    previous = runtime.collector.per_source
    runtime.collector.per_source = min(previous, remaining)
    try:
        terms = getattr(runtime, 'candidate_names', {}).get(owner)
        return runtime.collector.collect(url, owner, role, title=title, pages=pages, required_terms=terms)
    finally:
        runtime.collector.per_source = previous


def _collect_results(runtime, query, owner, role, domains=None, page_limit=3):
    started = runtime.collector.page_count
    if page_limit <= 0 or started >= runtime.config['max_pages'] or not _within_time(runtime):
        return 0
    previous = [q for q in runtime.discovery['queries'] if q['owner'] == owner]
    if any(q['query'] == query for q in previous):
        return 0
    if not owner.startswith('__') and len(previous) >= runtime.config.get('company_search_limit', 5):
        return 0
    record = {'query': query, 'owner': owner, 'role': role, 'results': 0, 'new_pages': 0}
    runtime.discovery['queries'].append(record)
    try:
        results = runtime.search(query, domains)
        record['results'] = len(results)
    except Exception as error:
        record['error'] = type(error).__name__
        runtime.collector._error('', f'검색 실패: {type(error).__name__}', 'search_failed')
        _diagnostics(runtime)
        return 0
    collected = set()
    for item in results:
        remaining = min(runtime.config['max_pages'] - runtime.collector.page_count,
                        page_limit - (runtime.collector.page_count - started))
        if remaining <= 0:
            break
        pages = _collect_url(runtime, item['url'], owner, role, remaining, title=item.get('title'))
        collected.update(page['source_id'] for page in pages if page['company_id'] == owner)
    record['new_pages'] = runtime.collector.page_count - started
    record['matched_pages'] = len(collected)
    _diagnostics(runtime)
    return len(collected)


def _add_local(runtime):
    manifest = ROOT / 'data/local_sources.json'
    for item in json.loads(manifest.read_text(encoding='utf-8')):
        owner = item.get('company_id') or company_id(item['company_name'])
        path = (ROOT / 'data' / item['file']).resolve()
        runtime.collector.add_local(path, owner, item['doc_type'], url=item.get('url', ''),
                                    title=item.get('title'), pages=item.get('pages'))


def _index(runtime):
    previous = runtime.corpus
    runtime.corpus = Corpus(runtime.collector.sources, runtime.config, ROOT / '.cache',
                           embeddings=previous.base_embeddings if previous else None)
    if previous:
        runtime.corpus.history = previous.history + runtime.corpus.history


def _literal_span(text, quote):
    normalized, positions = [], []
    for index, character in enumerate(text):
        for unit in compact(character):
            normalized.append(unit)
            positions.append(index)
    needle = compact(quote)
    start = ''.join(normalized).find(needle) if needle else -1
    return (positions[start], positions[start + len(needle) - 1] + 1) if start >= 0 else None


def _candidate_evidence(hint, sources, other_names=(), max_distance=500):
    """기업 제목과 바로 뒤 설명을 원문 그대로 연결하며 다른 기업 구간은 넘지 않습니다."""
    known = {s['source_id']: s for s in sources}
    names = [hint['name']] + hint['aliases']
    result = []
    for evidence in hint['evidence']:
        source = known.get(evidence['source_id'])
        if not source:
            continue
        text = source['text']
        span = _literal_span(text, evidence['quote'])
        if not span:
            continue
        start, end = span
        if _mentions_company(evidence['quote'], names) and len(compact(evidence['quote'])) >= 8:
            result.append({'source_id': source['source_id'], 'quote': text[start:end]})
            continue
        prefix = text[max(0, start - max_distance):start]
        previous = None
        page = source.get('rendered_page') or source.get('original_page')
        if start < max_distance and page:
            previous = next((s for s in sources if s.get('url') == source.get('url') and
                             (s.get('rendered_page') or s.get('original_page')) == page - 1), None)
        prior_text = previous['text'][-max_distance:] if previous else ''
        combined = (prior_text + '\n' if previous else '') + prefix
        headings = [(m.start(), m.end()) for m in re.finditer(r'[^\n]+', combined)
                    if any(compact(m.group()).casefold() == compact(name).casefold() for name in names)]
        if not headings:
            continue
        heading_start, heading_end = headings[-1]
        between = combined[heading_end:]
        if len(between) > max_distance or re.search(r'상세내용\s*닫기|</article>', between):
            continue
        other = [name for name in other_names if not any(compact(name).casefold() == compact(n).casefold() for n in names)]
        if any(compact(line).casefold() == compact(name).casefold() for line in between.splitlines() for name in other):
            continue
        previous_length = len(prior_text) + 1 if previous else 0
        if heading_start >= previous_length:
            raw_start = max(0, start - max_distance) + heading_start - previous_length
            result.append({'source_id': source['source_id'], 'quote': text[raw_start:end]})
        elif previous:
            # PDF 쪽 경계를 넘으면 두 쪽 각각의 연속 원문을 인용합니다.
            prior_start = len(previous['text']) - len(prior_text) + heading_start
            prior_start = max(0, prior_start - 60) if len(compact(previous['text'][prior_start:])) < 8 else prior_start
            result.extend([{'source_id': previous['source_id'], 'quote': previous['text'][prior_start:]},
                           {'source_id': source['source_id'], 'quote': text[:end]}])
    return list({(e['source_id'], e['quote']): e for e in result}.values())


def _has_domain_description(evidence, names):
    text = ' '.join(e['quote'] for e in evidence)
    text = re.sub(r'https?://\S+', '', text)
    for name in sorted(names, key=len, reverse=True):
        text = re.sub(re.escape(name), '', text, flags=re.I)
    return bool(DOMAIN_WORDS.search(text) and re.search(
        r'개발|솔루션|기술|제어|인지|자율|자동화|시스템|센서|공급|develop|solution|software|autonomous|control|delivery|sensor|warehouse|manufactur', text, re.I))


def _extract_batches(runtime, request, candidates, seen, processed, raw_batches, target):
    sources = [s for s in runtime.collector.sources
               if s['company_id'] == '__discovery__' and s['source_id'] not in processed]
    size = runtime.config.get('candidate_extract_batch_pages', 4)
    for offset in range(0, len(sources), size):
        if len(candidates) >= target or not _within_time(runtime):
            break
        if len(raw_batches) >= runtime.config.get('candidate_extract_max_calls', 16):
            runtime.discovery['stop_reason'] = 'extraction_call_limit'
            break
        batch = sources[offset:offset + size]
        processed.update(s['source_id'] for s in batch)
        record = {'source_ids': [s['source_id'] for s in batch], 'companies': [], 'rejected': []}
        raw_batches.append(record)
        try:
            answer = runtime.ask(CandidateHints,
                '각 자료 페이지를 빠짐없이 읽고 국내 Physical AI 후보 기업을 최대 8개 추출합니다. '
                '로봇의 인지·제어, 자율주행·운항, 무인 이동 시스템과 직접 관련된 기업만 후보입니다. '
                '의료 영상 판독만 하는 AI, 일반 소프트웨어, 투자사·기관·대기업·제품명은 제외합니다. '
                '상장·인수·폐업 사실이 본문에 명시된 기업도 제외합니다. 현재 자격은 후속 단계에서 확인합니다. '
                '기업명이 제목에 있고 설명이 아래에 있으면 기업명 제목 근거와 기술 설명 근거를 별도 인용합니다. '
                '가능하면 기업명부터 설명까지 같은 기업 구간의 연속 원문을 그대로 인용합니다. '
                '회사명에 Robotics가 있다는 사실만으로 사업을 추정하지 않습니다. 설명이 없으면 제외합니다. '
                'overview는 실제 사업 설명만 요약하고 추정·알려져 있다는 표현으로 보충하지 않습니다. '
                '별칭·법인명·홈페이지는 본문에서 확인한 경우만 기록합니다. 목록을 채우려고 기업을 만들지 않습니다.',
                {'request': request, 'already_found': [c['name'] for c in candidates], 'sources': batch})
            record['companies'] = answer['companies']
        except Exception as error:
            record['error'] = type(error).__name__
            runtime.collector._error('', f'후보 추출 실패: {type(error).__name__}', 'candidate_extraction_failed')
        for hint in record['companies']:
            aliases_in_text = [name for name in hint['aliases'] if name.strip() and
                               any(_mentions_company(s['text'], [name]) for s in batch)]
            names = list(dict.fromkeys(n.strip() for n in [hint['name']] + aliases_in_text if n.strip()))
            evidence = _candidate_evidence({**hint, 'aliases': names[1:]}, batch,
                                           [c['name'] for c in record['companies']])
            aliases = {company_id(name) for name in names}
            reason = None
            if not names or aliases & seen:
                reason = 'empty_or_duplicate_name'
            elif not valid_evidence(evidence, batch):
                reason = 'invalid_company_quote'
            elif not _has_domain_description(evidence, names):
                reason = 'domain_not_in_quoted_evidence'
            elif re.search(r'추정|알려져\s*있|가능성이|보입니다|보임', hint['overview']):
                reason = 'speculative_overview'
            if reason:
                record['rejected'].append({'name': hint['name'], 'reason': reason})
                continue
            if len(candidates) >= target:
                record['rejected'].append({'name': hint['name'], 'reason': 'candidate_check_limit'})
                continue
            website = hint['website']
            if not website.startswith(('http://', 'https://')) or not any(website in s['text'] for s in batch):
                website = ''
            cid = company_id(names[0])
            candidate = {**hint, 'name': names[0], 'aliases': names[1:], 'website': website,
                         'company_id': cid, 'evidence': evidence,
                         'source_ids': list(dict.fromkeys(e['source_id'] for e in evidence))}
            candidates.append(candidate)
            seen.update(aliases)
        save_json(runtime.output_dir / 'candidate_extraction_raw.json', raw_batches)
        save_json(runtime.output_dir / 'candidate_hints.json', candidates)
        runtime.discovery['extraction_batches'] = len(raw_batches)
        runtime.discovery['extraction_pages'] = len(processed)
        runtime.discovery['candidate_count'] = len(candidates)
        _diagnostics(runtime)


def _own_sources(runtime, cid):
    return [s for s in runtime.collector.sources if s['company_id'] == cid]


def _prepare_company(runtime, candidate, quota):
    cid, name = candidate['company_id'], candidate['name']
    runtime.candidate_names[cid] = [name] + candidate['aliases']
    started = runtime.collector.page_count

    def remaining():
        return max(0, quota - (runtime.collector.page_count - started))

    website = candidate['website']
    if website:
        _collect_url(runtime, website, cid, 'company', min(2, remaining()))
        if remaining() > 0:
            for link in runtime.collector.links(website, limit=50):
                if remaining() <= 0:
                    break
                if urlsplit(link['url']).hostname != urlsplit(website).hostname:
                    continue
                if re.search(r'about|company|team|회사|기업|소개', link['url'] + link['title'], re.I):
                    _collect_url(runtime, link['url'], cid, 'company', min(1, remaining()), title=link['title'])
                    break
    # 자격 자료와 기술·시장 자료에 각각 공간을 남겨 놓습니다.
    queries = [('company', '기업 소개 대표 본사 투자 단계'),
               ('company', '상장 인수 폐업 영업 종료 최신'),
               ('technology', '기술 제품 특허 성능'),
               ('market', '고객 계약 도입 과금')]
    for index, (role, suffix) in enumerate(queries):
        allowance = min(2, max(0, remaining() - (len(queries) - index - 1)))
        _collect_results(runtime, f'"{name}" {suffix}', cid, role, page_limit=allowance)
    own = _own_sources(runtime, cid)
    missing = [role for role in ['technology', 'market']
               if not any(role in s.get('doc_types', [s['doc_type']]) or 'company' in s.get('doc_types', [s['doc_type']]) for s in own)]
    if remaining() and (not own or missing):
        alias = candidate['aliases'][0] if candidate['aliases'] else name
        suffix = '로봇 공식 홈페이지' if not own else ('제품 기술' if 'technology' in missing else '고객 도입 사례')
        _collect_results(runtime, f'"{alias}" {suffix}', cid, 'company', page_limit=min(2, remaining()))
    own = _own_sources(runtime, cid)
    counts = {role: sum(role in s.get('doc_types', [s['doc_type']]) or
                        (role != 'company' and 'company' in s.get('doc_types', [s['doc_type']])) for s in own)
              for role in ['company', 'technology', 'market']}
    runtime.discovery['coverage'][cid] = {'name': name, 'unique_pages': len(own), **counts}
    _diagnostics(runtime)
    return own


def _explicit_status(candidate, sources):
    """회사명이 포함된 연속 원문의 완료 사실만 보수적으로 감지합니다."""
    names = [candidate['name']] + candidate['aliases']
    events = []
    patterns = {
        'unlisted': re.compile(r'(?:코스닥|코스피|코넥스|KOSDAQ|KOSPI).{0,18}(?:상장법인|상장기업|상장사|등록했다|상장했다|상장했습니다|상장된|상장한|상장\s*완료)|(?:코스닥|코스피|코넥스)\s*(?:시장\s*)?상장(?=\s*(?:[.。\n]|$))', re.I | re.S),
        'operating': re.compile(r'폐업(?:한|했|하였|\s*완료|\s*\()|폐업\s*(?:상태|기업)|(?:^|\n)\s*폐업\s*(?:\n|$)', re.I),
    }
    for source in sources:
        text = source['text']
        title_matches = _mentions_company(source.get('title', ''), names)
        for field, pattern in patterns.items():
            for match in pattern.finditer(text):
                start, end = max(0, match.start() - 100), min(len(text), match.end() + 80)
                quote = text[start:end].strip()
                if not _mentions_company(quote, names) and not (title_matches and match.start() < 300 and source['company_id'] == candidate['company_id']):
                    continue
                # 같은 기사에 등장한 고객·투자사·경쟁사의 상장/폐업을 대상기업에 귀속시키지 않습니다.
                # 애매하면 자동 fail로 덮어쓰지 않고 자격 검수의 unknown을 유지합니다.
                before = text[max(0, match.start() - 180):match.start()]
                mentions = [m for name in names if name for m in re.finditer(re.escape(name), before, re.I)]
                if mentions:
                    tail = before[max(m.end() for m in mentions):]
                    subjects = re.findall(r'(?:^|[.!?]\s*)([가-힣A-Za-z][가-힣A-Za-z0-9·_-]{1,35})(?:은|는|이|가)\s', tail)
                    if any(subject not in {'회사', '기업', '법인', '당사', '현재', '이후'} for subject in subjects):
                        continue
                    if re.search(r'고객사|투자사|경쟁사|협력사|모회사|자회사', tail):
                        continue
                elif re.search(r'([가-힣A-Za-z][가-힣A-Za-z0-9·_-]{1,35})(?:은|는|이|가)\s', before):
                    continue
                if re.search(r'상장\s*(?:예정|계획|목표|준비)|상장예비심사|폐업률|폐업\s*(?:위기|방지)|(?:종료|중단)\s*(?:예정|계획)', quote):
                    continue
                if re.search(r'(?:상장기업|상장법인|상장사|폐업한).{0,15}(?:아니|않|미확인)', quote):
                    continue
                events.append({'field': field, 'status': 'fail', 'reason': '수집 원문에 상장 완료 사실이 명시되어 있습니다.' if field == 'unlisted' else '수집 원문에 폐업·영업 종료 사실이 명시되어 있습니다.',
                               'evidence': [{'source_id': source['source_id'], 'quote': quote}],
                               'published_at': source.get('published_at')})
                break
    return events


def _evidence_date(evidence, known):
    dates = []
    for item in evidence:
        source = known.get(item['source_id'], {})
        for year, month, day in re.findall(r'(20\d{2})[.년/-]\s*(\d{1,2})(?:[.월/-]\s*(\d{1,2}))?', item['quote']):
            dates.append((int(year), int(month), int(day or 1)))
        if not dates:
            match = re.match(r'(20\d{2})-(\d{2})-(\d{2})', source.get('published_at') or '')
            if match:
                dates.append(tuple(map(int, match.groups())))
    return max(dates) if dates else None


def _reject_absence_inference(eligibility):
    for key in ['unlisted', 'no_exit']:
        item = eligibility[key]
        if item['status'] != 'pass':
            continue
        reason = compact(item['reason']).casefold()
        absent = re.search(r'(?:정보|기사|공시|기록|근거|발표|사실|내용|내역).{0,12}(?:없|미확인|불명확|찾지못|발견되지|확인되지)|검색.{0,10}(?:없|안됨|되지않|못)', reason)
        speculative = re.search(r'추정|간주|보임|보입니다', reason)
        if absent or speculative:
            label = '비상장' if key == 'unlisted' else '인수 Exit 미완료'
            item.update(_unknown(f'{label}를 확인할 직접 근거가 필요합니다. 검색 결과의 부재나 추정으로 확정하지 않습니다.'))


def _qualify(runtime, candidate):
    # 역할 구분과 무관하게 법인·상장·폐업 근거를 모두 검토합니다.
    chunks = runtime.corpus.evidence_for(
        [f'{candidate["name"]} 법인 대표 본사 사업 투자 상장 폐업 인수'], candidate['company_id'], None)
    own = _own_sources(runtime, candidate['company_id'])
    initial = [s for s in runtime.collector.sources if s['source_id'] in candidate['source_ids']]
    sources = list({s['source_id']: s for s in own + runtime.corpus.source_pages(chunks) + initial}.values())
    explicit = _explicit_status(candidate, sources)
    try:
        result = runtime.ask(EligibilityResult,
            '초기 후보와 후속 원문의 법인명·대표자·사업·투자자를 먼저 대조합니다. '
            '동명이거나 사업 설명이 상충하면 identity=unknown이며 국내·Physical AI 여부도 unknown입니다. '
            '동일 초기 기사만 다시 인용하거나 이름만 같은 것은 동일 법인 확인이 아닙니다. '
            '국내 법인, 현재 비상장, 최신 Seed~Series C, 인수 Exit 미완료, 로봇의 인지·제어 또는 자율 이동 사업을 확인합니다. '
            '의료영상 판독·일반 소프트웨어만 제공하면 physical_ai=fail입니다. 단순 3D 제조와 로봇 제어를 혼동하지 않습니다. '
            '공식 자료의 상장 완료·코스닥 등록과 상장 준비를 구분합니다. 검색 결과가 없다는 이유로 비상장·미인수를 pass하지 않습니다. '
            '비공개 투자 금액을 비상장 증거로 쓰지 않습니다. 운영 상태 operating은 현재 영업 지속 근거가 있으면 pass, 폐업·사업 종료 완료면 fail, 미확인이면 unknown입니다. '
            '과거 창업·투자 발표만으로 현재 영업을 확정하지 않습니다. 원문의 날짜를 비교해 최신 상태를 반영하고, 폐업 이후 재개가 확인되지 않으면 과거 영업 자료로 pass하지 않습니다. '
            '판정별 직접 근거와 출처를 제시하고 애매한 항목을 추정하지 않습니다.',
            {'company': candidate['name'], 'initial_overview': candidate['overview'],
             'initial_evidence': candidate['evidence'], 'checked_at': datetime.now(timezone.utc).date().isoformat(),
             'explicit_events': explicit, 'sources': sources})
        identity, eligibility = result['identity'], result['eligibility']
        operating = result.get('operating', _unknown('현재 영업 상태를 확인해야 합니다.'))
    except Exception as error:
        reason = f'자격 확인 응답 실패({type(error).__name__}): 기업 자격을 확인해야 합니다.'
        eligibility, identity, operating = _unknown_eligibility(reason), _unknown(reason), _unknown(reason)
        runtime.collector._error('', f'{candidate["name"]}: {reason}', 'eligibility_failed')
    _reject_absence_inference(eligibility)
    items = [identity] + [eligibility[k] for k in QUALIFICATION_KEYS[1:6]] + [operating]
    known = {s['source_id']: s for s in sources}
    names = [candidate['name']] + candidate['aliases']
    review = []
    for number, item in enumerate(items, 1):
        anchored = []
        verified_contexts = set()
        for evidence in item['evidence']:
            source = known.get(evidence['source_id'], {})
            if source.get('company_id') == '__discovery__' and not _mentions_company(evidence['quote'], names):
                expanded = _candidate_evidence({'name': candidate['name'], 'aliases': candidate['aliases'],
                                                'evidence': [evidence]}, sources, max_distance=1200)
                verified_contexts.update((e['source_id'], e['quote']) for e in expanded)
                anchored.extend(expanded or [evidence])
            else:
                anchored.append(evidence)
        item['evidence'] = anchored
        scoped = all(e['source_id'] in known and
                     (known[e['source_id']]['company_id'] == candidate['company_id'] or
                      _mentions_company(e['quote'], names) or
                      (e['source_id'], e['quote']) in verified_contexts) for e in item['evidence'])
        if item['status'] != 'unknown' and not (scoped and valid_evidence(item['evidence'], sources)):
            item.update(_unknown('등록된 원문에서 자격 판단의 인용을 확인하지 못했습니다.'))
        if number == 1 and item['status'] == 'pass' and not any(known[e['source_id']]['company_id'] == candidate['company_id'] for e in item['evidence']):
            item.update(_unknown('초기 후보 기사 이외의 자료로 동일 법인을 확인해야 합니다.'))
        if item['status'] != 'unknown':
            review.append({'criterion_id': number, 'field': QUALIFICATION_KEYS[number - 1], **item})
    if review:
        try:
            checked = runtime.ask(EvidenceReview,
                '기업 자격 판단을 독립 검수합니다. 각 field의 status가 인용문과 전체 자료에서 직접 확인될 때만 supported=true입니다. '
                'supported는 기업의 자격 충족 여부가 아니라 입력 status 판단의 근거 일치 여부입니다. '
                '예: domestic=fail이고 미국 본사 원문이 확인되면 supported=true입니다. 국내 기업이 아니라는 이유로 false를 주면 안 됩니다. '
                '예: unlisted=fail이고 코스닥 상장 완료 원문이 있으면 supported=true입니다. '
                '예: no_exit=pass이고 인수 기사를 찾지 못했다는 이유뿐이면 supported=false입니다. '
                '초기/후속 자료의 사업·대표자 불일치를 무시한 동일 법인 확정, 의료 영상 AI를 로봇 제어로 해석, '
                '상장 또는 인수 기사가 없다는 이유로 비상장·미인수 확정, 비공개 투자금을 비상장으로 해석하면 false입니다. '
                '초기 기사 자체는 동일 기업 후속 확인이 아닙니다. 자료가 서로 상충하면 false입니다. '
                '과거 창업·투자 뉴스만으로 현재 영업 지속을 확정하지 않습니다. 폐업 완료보다 오래된 자료는 현재 영업 근거가 아닙니다. '
                '모든 입력 criterion_id를 한 번씩 검수하고, 불확실하면 false로 판정합니다.',
                {'company': candidate['name'], 'initial_overview': candidate['overview'],
                 'initial_evidence': candidate['evidence'], 'findings': review, 'sources': sources})
            checks = {}
            for check in checked['checks']:
                checks.setdefault(check['criterion_id'], []).append(check)
            for record in review:
                values = checks.get(record['criterion_id'], [])
                if len(values) != 1 or not values[0]['supported']:
                    items[record['criterion_id'] - 1].update(_unknown('자격 근거 검수: ' + (values[0]['reason'] if len(values) == 1 else '검수 결과 누락·중복')))
        except Exception as error:
            for record in review:
                items[record['criterion_id'] - 1].update(_unknown(f'자격 근거 검수 실패({type(error).__name__})'))
    for event in explicit:
        # 명시적인 완료 사실을 모델의 근거 없는 pass가 덮어쓰지 못합니다.
        item = eligibility['unlisted'] if event['field'] == 'unlisted' else operating
        if event['field'] == 'operating' and operating['status'] == 'pass' and any(
                re.search(r'(?:영업|운영|사업)\s*(?:을\s*)?재개|폐업\s*취소', e['quote']) for e in operating['evidence']):
            restart, closure = _evidence_date(operating['evidence'], known), _evidence_date(event['evidence'], known)
            if restart and closure and restart > closure:
                continue
            if not restart or not closure:
                operating.update(_unknown('폐업 기록과 영업 재개 기록의 날짜를 비교해야 합니다.'))
                continue
        item.update({k: event[k] for k in ['status', 'reason', 'evidence']})
    if identity['status'] != 'pass':
        for key in ['domestic', 'physical_ai']:
            eligibility[key] = _unknown(identity['reason'])
    candidate.update(identity=identity, eligibility=eligibility, operating=operating, sources=sources)
    return candidate


def requalify(runtime, candidates):
    """이미 저장한 원문으로 자격만 다시 검수합니다. 웹 검색·자료 수집은 하지 않습니다."""
    runtime.collector = Collector(runtime.config, runtime.output_dir / 'corpus')
    runtime.candidate_names = {c['company_id']: [c['name']] + c.get('aliases', []) for c in candidates}
    _index(runtime)
    previous_unresolved = runtime.output_dir / 'unresolved_candidates.json'
    accepted, excluded = [], []
    unresolved = json.loads(previous_unresolved.read_text(encoding='utf-8')) if previous_unresolved.exists() else []
    known = {s['source_id']: s for s in runtime.collector.sources}
    for original in candidates:
        candidate = {k: v for k, v in original.items() if k not in {'identity', 'eligibility', 'operating', 'sources'}}
        initial = [known[sid] for sid in candidate.get('source_ids', []) if sid in known and known[sid]['company_id'] == '__discovery__']
        if not valid_evidence(candidate.get('evidence', []), initial) or not any(
                _mentions_company(e['quote'], runtime.candidate_names[candidate['company_id']]) for e in candidate.get('evidence', [])):
            candidate.update(data_status='invalid_initial_evidence', reason='초기 후보의 기업명·원문 인용을 확인해야 합니다.')
            unresolved.append(candidate)
        elif not _own_sources(runtime, candidate['company_id']):
            candidate.update(data_status='unavailable', reason='기업별 공개 원문을 확보하지 못했습니다.')
            unresolved.append(candidate)
        else:
            candidate = _qualify(runtime, candidate)
            (excluded if candidate['operating']['status'] == 'fail' or
             any(item['status'] == 'fail' for item in candidate['eligibility'].values()) else accepted).append(candidate)
        for filename, value in [('candidates.json', accepted), ('excluded_candidates.json', excluded),
                                ('unresolved_candidates.json', unresolved)]:
            save_json(runtime.output_dir / filename, value)
    corpus_pdf = runtime.collector.export_pdf()
    save_json(runtime.output_dir / 'retrieval_log.json', runtime.corpus.history)
    summary = {'checked': len(candidates), 'accepted': len(accepted), 'excluded': len(excluded),
               'unresolved': len(unresolved), 'page_count': runtime.collector.page_count,
               'web_searches': 0, 'new_pages': 0}
    save_json(runtime.output_dir / 'requalification.json', summary)
    return accepted, excluded


def run(runtime, request):
    config = runtime.config
    runtime.preparation_started = time.monotonic()
    runtime.discovery = {'target': config['max_candidates'], 'queries': [], 'rounds': [], 'coverage': {}, 'stop_reason': None}
    runtime.candidate_names = {}
    collector = runtime.collector = Collector(config, runtime.output_dir / 'corpus')
    _add_local(runtime)
    registry = json.loads((ROOT / 'config/sources.json').read_text(encoding='utf-8'))
    for seed in registry['seed_urls']:
        seed = {'url': seed} if isinstance(seed, str) else seed
        if any(s['company_id'] == '__discovery__' and s.get('url') == seed['url'] for s in collector.sources):
            continue
        remaining = config['seed_page_budget'] - sum(s['company_id'] == '__discovery__' for s in collector.sources)
        if remaining <= 0:
            break
        _collect_url(runtime, seed['url'], '__discovery__', 'company', min(seed.get('max_pages', 4), remaining),
                     title=seed.get('title'), pages=seed.get('pages'))
    common_start = collector.page_count
    for item in registry.get('common_sources', []):
        if any(s['company_id'] == '__sector__' and s.get('url') == item['url'] for s in collector.sources):
            continue
        remaining = config.get('common_page_budget', 8) - (collector.page_count - common_start)
        if remaining <= 0:
            break
        _collect_url(runtime, item['url'], '__sector__', 'market', min(item.get('max_pages', 4), remaining),
                     title=item.get('title'), pages=item.get('pages'))
    candidates, accepted, excluded, unresolved = [], [], [], []
    seen, processed, raw_batches, cursor = set(), set(), [], 0
    queries = registry.get('discovery_queries') or [{'query': request + ' 스타트업 투자 기업'}]
    maximum_rounds = min(len(queries), config.get('candidate_search_rounds', 6))
    check_limit = config.get('candidate_check_limit', max(20, config['max_candidates']))
    no_new = 0
    for round_no in range(maximum_rounds + 1):
        before = len(candidates)
        if round_no:
            item = queries[round_no - 1]
            remaining = config['seed_page_budget'] - sum(s['company_id'] == '__discovery__' for s in collector.sources)
            _collect_results(runtime, item['query'], '__discovery__', 'company', item.get('domains'), min(4, remaining))
        _extract_batches(runtime, request, candidates, seen, processed, raw_batches, check_limit)
        added = len(candidates) - before
        no_new = no_new + 1 if added == 0 else 0
        runtime.discovery['rounds'].append({'round': round_no, 'new_candidates': added, 'total': len(candidates)})
        while cursor < len(candidates) and len(accepted) < config['max_candidates'] and _within_time(runtime):
            candidate = candidates[cursor]
            cursor += 1
            reserve = max(0, config['seed_page_budget'] - sum(s['company_id'] == '__discovery__' for s in collector.sources))
            available = max(0, config['max_pages'] - collector.page_count - reserve)
            quota = min(config.get('company_page_budget', 9), available // max(1, config['max_candidates'] - len(accepted)))
            if quota <= 0:
                runtime.discovery['stop_reason'] = 'company_page_budget_exhausted'
                cursor -= 1
                break
            own = _prepare_company(runtime, candidate, quota)
            if not own:
                candidate['data_status'] = 'unavailable'
                candidate['reason'] = '공개 원문을 확보하지 못했습니다. 수집 실패와 기업 부적격은 구분합니다.'
                unresolved.append(candidate)
                continue
            _index(runtime)
            candidate = _qualify(runtime, candidate)
            (excluded if candidate['operating']['status'] == 'fail' or
             any(x['status'] == 'fail' for x in candidate['eligibility'].values()) else accepted).append(candidate)
            save_json(runtime.output_dir / 'candidates.json', accepted)
            save_json(runtime.output_dir / 'excluded_candidates.json', excluded)
            save_json(runtime.output_dir / 'unresolved_candidates.json', unresolved)
        _diagnostics(runtime)
        if len(accepted) >= config['max_candidates']:
            runtime.discovery['stop_reason'] = 'candidate_target_reached'
            break
        if not _within_time(runtime) or runtime.discovery['stop_reason'] == 'company_page_budget_exhausted':
            break
        if no_new >= config.get('candidate_no_new_rounds', 3):
            runtime.discovery['stop_reason'] = 'no_new_candidates'
            break
        if cursor >= check_limit:
            runtime.discovery['stop_reason'] = 'candidate_check_limit'
            break
    runtime.discovery['stop_reason'] = runtime.discovery['stop_reason'] or 'discovery_round_limit'
    runtime.discovery.update(accepted=len(accepted), excluded=len(excluded), unresolved=len(unresolved), checked=cursor,
                             page_count=collector.page_count, seconds=round(time.monotonic() - runtime.preparation_started, 2))
    for name, data in [('candidate_hints.json', candidates), ('candidates.json', accepted),
                       ('excluded_candidates.json', excluded), ('unresolved_candidates.json', unresolved)]:
        save_json(runtime.output_dir / name, data)
    if not collector.sources:
        raise ValueError('수집된 자료가 없습니다. corpus/collection_errors.json과 data/local_sources.json을 확인해 주세요.')
    if runtime.corpus is None:
        _index(runtime)
    corpus_pdf = collector.export_pdf()
    save_json(runtime.output_dir / 'retrieval_log.json', runtime.corpus.history)
    _diagnostics(runtime)
    return {'candidate_list': accepted, 'doc_pages': collector.page_count, 'page_cap': 200,
            'prepared': {'corpus': str(runtime.output_dir / 'corpus'), 'corpus_pdf': str(corpus_pdf),
                         'excluded': excluded, 'unresolved': unresolved,
                         'discovery': runtime.discovery, 'collection_errors': collector.errors}}
```
